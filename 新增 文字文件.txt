<script>
const code=[
'public boolean isSymmetric(TreeNode root) {',
'    if (root == null) return true;',
'    return isMirror(root.left, root.right);',
'}',
'',
'private boolean isMirror(TreeNode left, TreeNode right) {',
'    if (left == null && right == null) return true;',
'    if (left == null || right == null) return false;',
'    if (left.val != right.val) return false;',
'    return isMirror(left.left, right.right)',
'        && isMirror(left.right, right.left);',
'}'
];
const pos={root:[280,65],left:[180,150],right:[380,150],leftLeft:[120,268],leftRight:[230,268],rightLeft:[330,268],rightRight:[440,268]};
const steps=[
{line:0,label:'主方法入口',active:['root'],text:'LeetCode 呼叫這個方法，把整棵樹 root 傳進來。',formula:'目標：判斷 root 是否左右對稱'},
{line:1,label:'空樹判斷',active:[],text:'如果 root 是 null，空樹沒有不對稱的地方，所以回傳 true。',formula:'root == null → true'},
{line:2,label:'開始鏡像比較',active:['left','right'],pair:['left','right'],text:'真正要比的是 root 的左子樹和右子樹。',formula:'isMirror(root.left, root.right)'},
{line:5,label:'helper 的任務',active:['left','right'],pair:['left','right'],text:'isMirror 每次只負責比較一對節點：left 和 right 是否互為鏡像。',formula:'left vs right'},
{line:6,label:'兩邊都空',active:[],text:'兩邊都是 null，代表這一對位置都沒有節點，是對稱的。',formula:'left == null && right == null → true'},
{line:7,label:'只有一邊空',active:['left'],text:'如果只有一邊是 null，代表結構不一樣，直接 false。',formula:'left == null || right == null → false'},
{line:8,label:'比較目前值',active:['left','right'],pair:['left','right'],text:'兩邊都不是 null 後，才能比較 left.val 和 right.val。值不同就不是鏡像。',formula:'left.val != right.val → false'},
{line:9,label:'比較外側',active:['leftLeft','rightRight'],done:['left','right'],pair:['leftLeft','rightRight'],text:'外側對外側：left.left 要對上 right.right。',formula:'isMirror(left.left, right.right)'},
{line:10,label:'比較內側',active:['leftRight','rightLeft'],done:['left','right','leftLeft','rightRight'],pair:['leftRight','rightLeft'],text:'內側對內側：left.right 要對上 right.left。',formula:'isMirror(left.right, right.left)'},
{line:9,label:'合併結果',active:['left','right'],done:['leftLeft','rightRight','leftRight','rightLeft'],pair:['left','right'],text:'外側成立，而且內側也成立，這一對才算鏡像，所以用 &&。',formula:'外側鏡像 && 內側鏡像'}
];
let i=0;
const qs=s=>document.querySelector(s);
const nodes=new Map([...document.querySelectorAll('.node')].map(n=>[n.dataset.node,n]));
function esc(s){return s.replaceAll('&','&amp;').replaceAll('<','&lt;').replaceAll('>','&gt;')}
function render(){
  const s=steps[i];
  qs('#code').innerHTML=code.map((l,idx)=>`<span class="line${idx===s.line?' now':''}">${esc(l)}</span>`).join('');
  nodes.forEach(n=>n.classList.remove('active','done'));
  (s.done||[]).forEach(n=>nodes.get(n).classList.add('done'));
  s.active.forEach(n=>nodes.get(n).classList.add('active'));
  const line=qs('#line');
  if(s.pair){const[a,b]=s.pair;line.setAttribute('x1',pos[a][0]);line.setAttribute('y1',pos[a][1]);line.setAttribute('x2',pos[b][0]);line.setAttribute('y2',pos[b][1]);line.classList.add('show')}
  else line.classList.remove('show');
  qs('#label').textContent=s.label; qs('#text').textContent=s.text; qs('#formula').textContent=s.formula;
  qs('#counter').textContent=`${i+1} / ${steps.length}`;
  qs('#prev').disabled=i===0; qs('#next').disabled=i===steps.length-1;
}
qs('#prev').onclick=()=>{i=Math.max(0,i-1);render()};
qs('#next').onclick=()=>{i=Math.min(steps.length-1,i+1);render()};
qs('#reset').onclick=()=>{i=0;render()};
render();
</script>
